# client fields with no Terraform attribute expressing them, reviewed and
# accepted. Maintained by hand: scripts/tf-field-coverage.ts reads the
# non-comment lines and never writes this file
# (fogpipe/cloud-workspace#888), so only add to it deliberately, on review,
# with the reason beside the entry, in the group it belongs to.

App.service_account_id
# The permission trio is `read`/`write`/`owner` in HCL; `can_` reads as a
# question, and a schema attribute is an assertion.
BucketKey.can_owner
BucketKey.can_read
BucketKey.can_write
# The binding names its member as a type and an id, which is what the API
# takes. `member` is the joined form the CLI prints.
IAMBinding.member
# The resource this binds to is `project_id` — the provider has one binding
# resource per scope rather than one carrying a discriminator.
IAMBinding.resource_id
IAMBinding.resource_type

# --- belongs to another resource ----------------------------------------
# A domain is `fpcloud_domain`, and a domain attached to a bucket is that
# resource's own. Reading them back onto the app would make one hostname two
# owners in one state file.
App.domains
Domain.bucket_id
# A job's app is `app`, resolved by name like every other reference.
Job.app_id
Job.app_name

# --- deploy-time state, not configuration -------------------------------
# What is running right now is the outcome of a deploy, which Terraform does
# not perform — declaring it would make every push show as drift.
App.release
Bucket.website_version
Job.last_run
Runner.message
Runner.problems
# A pool's live failures (#851) are a diagnostic that changes while nothing
# in the config does; `fpcloud runner show` and the console are where they
# belong.
Runner.instances
# Which runners are up and what job each is serving is the same kind of
# reading one level finer: it changes with GitHub's queue rather than with
# anything declared, so an attribute over it is drift on every plan.
Runner.queue
# GitHub's view of the pool's demand (#146): assigned, running and waiting
# jobs, read live from the listener. The same reading as instances, from the
# other side of the queue.
# The recycle record (#145): asked for, in progress, finished. It is an
# operation with a lifetime, not a state a config can hold — `terraform apply`
# has no way to say "restart now and not again", and an attribute over the
# stamp would either re-restart on every apply or be permanent drift. The
# restart is `fpcloud runner restart`.
Runner.restart
# What a bucket currently holds is a measurement of the data plane, which the
# tenant writes to without Terraform: an attribute over it would move on every
# refresh and never converge on anything a config could state.
Bucket.object_count
Bucket.used_bytes
# How far the replica trails the primary right now (#300), read live from
# `pg_stat_replication` at request time. The same kind of reading as
# `Bucket.used_bytes` one layer down: it moves every second, driven by the
# tenant's own write volume rather than by anything declared, so an attribute
# over it would rewrite state on every refresh and converge on nothing. It is
# also the one field here a config could actively misuse — fed into another
# resource's argument it would replace that resource continuously. `fpcloud db
# show` and the console are where a staleness reading belongs.
Database.replication_lag_seconds

# --- the same state, spelled declaratively somewhere else ---------------
# `backup.enabled` is where a database says whether it is backed up. This is
# the server echoing that decision back inverted, so an attribute for it
# would be a second way to say one thing, and the two could disagree.
Database.backups_disabled

Bucket.garage_bucket_id
Project.namespace
# The parent is already the resource's own reference, so the id it echoes
# adds nothing to configure.
Bucket.project_id
Job.project_id
Runner.project_id

# --- an org member is invited, and the invitation is not the person -----
OrgMember.invited_by

# The secret's names are listed by the API for display; the values are

# An SA key's lifetime is fixed at creation and the API returns no way to
# change it; exposing the expiry alone would suggest otherwise.
ServiceAccountKey.expires_at

# --- a catalog instance is made, never declared (ADR-235) ------------------
# `template` names the catalog entry an app was deployed from by the console or
# the CLI, and `template_update` is the catalog's newer version on offer; an app
# declared in HCL is the practitioner's own image and carries neither, and the
# upgrade they point at is an action, not an attribute.
App.template
App.template_update

# --- project secrets (#1069) ----------------------------------------------
# Which apps mount a database's owner secret is live cluster-derived state a
# read reports, not configuration: fpcloud_app's secret_mounts is where the
# binding is declared, and expressing the reverse index as an attribute would
# put the same fact in two places of one plan.
Database.mounted_by
# Set only on a database's owner secret, which the platform creates and
# deletes with the database — never on one Terraform manages, so a config
# could never legitimately write it.
ProjectSecret.owner_database_id
